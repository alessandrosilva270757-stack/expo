import React, { useState } from "react";
import { View, Text, TextInput, TouchableOpacity, StyleSheet } from "react-native";

export default function App() {
  const [screen, setScreen] = useState("login");
  const [role, setRole] = useState("");

  const handleLogin = (type) => {
    setRole(type);
    if (type === "empresa") setScreen("empresa");
    if (type === "motoboy") setScreen("motoboy");
    if (type === "admin") setScreen("admin");
  };

  if (screen === "login") {
    return (
      <View style={styles.container}>
        <Text style={styles.title}>MOTOFAST</Text>
        <TouchableOpacity style={styles.button} onPress={() => handleLogin("empresa")}>
          <Text style={styles.buttonText}>Entrar como Empresa</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.button} onPress={() => handleLogin("motoboy")}>
          <Text style={styles.buttonText}>Entrar como Motoboy</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.button} onPress={() => handleLogin("admin")}>
          <Text style={styles.buttonText}>Entrar como Admin</Text>
        </TouchableOpacity>
      </View>
    );
  }

  if (screen === "empresa") {
    const [km, setKm] = useState("");
    const [corrida, setCorrida] = useState(null);

    const criarCorrida = () => {
      let valorEmpresa = 23 * parseFloat(km);
      if (km < 1) valorEmpresa = 11.5;
      setCorrida({ km, valorEmpresa });
    };

    return (
      <View style={styles.container}>
        <Text style={styles.title}>Painel Empresa</Text>
        <TextInput
          style={styles.input}
          placeholder="Distância em km"
          keyboardType="numeric"
          value={km}
          onChangeText={setKm}
        />
        <TouchableOpacity style={styles.button} onPress={criarCorrida}>
          <Text style={styles.buttonText}>Criar Corrida</Text>
        </TouchableOpacity>
        {corrida && (
          <Text style={styles.info}>
            Corrida criada: {corrida.km} km - Empresa paga R$ {corrida.valorEmpresa.toFixed(2)}
          </Text>
        )}
      </View>
    );
  }

  if (screen === "motoboy") {
    return (
      <View style={styles.container}>
        <Text style={styles.title}>Painel Motoboy</Text>
        <Text style={styles.info}>Aqui aparecerão as corridas disponíveis.</Text>
      </View>
    );
  }

  if (screen === "admin") {
    return (
      <View style={styles.container}>
        <Text style={styles.title}>Painel Admin</Text>
        <Text style={styles.info}>Admin visualiza todas as corridas aqui.</Text>
      </View>
    );
  }

  return null;
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: "#000",
    alignItems: "center",
    justifyContent: "center",
    padding: 20,
  },
  title: {
    fontSize: 28,
    fontWeight: "bold",
    marginBottom: 20,
    color: "#FFD700",
  },
  input: {
    borderWidth: 1,
    borderColor: "#FFD700",
    padding: 10,
    marginBottom: 10,
    width: "100%",
    borderRadius: 8,
    color: "#fff",
  },
  button: {
    backgroundColor: "#FFD700",
    padding: 12,
    borderRadius: 8,
    marginVertical: 5,
    width: "100%",
    alignItems: "center",
  },
  buttonText: {
    color: "#000",
    fontWeight: "bold",
  },
  info: {
    marginTop: 20,
    fontSize: 16,
    color: "#fff",
  },
});
